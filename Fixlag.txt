local function createTween(obj, props, time, style, dir)
	return TweenService:Create(obj, TweenInfo.new(time or 0.25, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props)
end

local function tweenAndPlay(obj, props, time, style, dir)
	local t = createTween(obj, props, time, style, dir)
	t:Play()
	return t
end

local function updateResponsiveScale()
	local camera = workspace.CurrentCamera
	if not camera then return end
	local viewport = camera.ViewportSize
	local baseWidth = 520
	local baseHeight = 400
	local padding = 40
	local scaleX = (viewport.X - padding) / baseWidth
	local scaleY = (viewport.Y - padding) / baseHeight
	local scale = math.min(scaleX, scaleY)
	scale = math.clamp(scale, 0.55, 1.15)
	panelScale.Scale = scale
end

	toastTitle.Text = titleText

local function shakePanel()
	local originalPos = mainPanel.Position
	local shakeOffsets = { 8, -8, 6, -6, 4, -4, 2, -2, 0 }
	task.spawn(function()
		for _, offset in ipairs(shakeOffsets) do
			if not mainPanel or not mainPanel.Parent then break end
			mainPanel.Position = UDim2.new(0.5, offset, 0.5, 0)
			task.wait(0.03)
		end
		mainPanel.Position = originalPos
	end)
end

local function validateToken(token)
	if typeof(token) ~= "string" then return false end
	local trimmed = token:gsub("^%s+", ""):gsub("%s+$", "")
	if #trimmed == 0 then return false end
	return trimmed == Geeow92JeIw815Neys752Mehska917.VOw91ane91Lwdi926Mevs85aneToken
end

local function performActivation(token)
	if isChecking or isActivated then return end
	isChecking = true
	setButtonState("checking")
	setStatus("[Checking]", THEME.checking, ASSETS.loading, true)

	task.delay(1.0, function()
		local valid = false
		pcall(function()
			valid = validateToken(token)
		end)

		if valid then
			setStatus("[Success]", THEME.success, ASSETS.check, false)
			setButtonState("success")
			saveToken(token)
			task.delay(0.4, function()
				showSuccessAnimation()
			end)
			task.delay(1.4, function()
				isActivated = true
				isChecking = false
				showToast("STRUGLIX", "Activating script...", THEME.success, ASSETS.loading)
				tweenAndPlay(mainPanel, { BackgroundTransparency = 1, Size = UDim2.fromOffset(480, 370) }, 0.4, Enum.EasingStyle.Quart, Enum.EasingDirection.In)
				tweenAndPlay(panelStroke, { Transparency = 1 }, 0.35)
				tweenAndPlay(accentBar, { BackgroundTransparency = 1 }, 0.35)
				tweenAndPlay(dimOverlay, { BackgroundTransparency = 1 }, 0.45)
				task.delay(0.5, function()
					screenGui.Enabled = false
					onActivationSuccess()
				end)
			end)
		else
			isChecking = false
			setStatus("[Invalid]", THEME.invalid, ASSETS.alert, false)
			setButtonState("retry")
			tweenAndPlay(tokenStroke, { Color = THEME.invalid, Transparency = 0 }, 0.2)
			tweenAndPlay(tokenBox, { BackgroundColor3 = Color3.fromRGB(30, 16, 18) }, 0.2)
			shakePanel()
			task.delay(1.6, function()
				if not isActivated then
					tweenAndPlay(tokenStroke, { Color = THEME.border }, 0.4)
					tweenAndPlay(tokenBox, { BackgroundColor3 = THEME.panelAlt }, 0.4)
				end
			end)
		end
	end)
end

local function tryAutoActivate()
	local saved = loadSavedToken()
	if saved then
		local valid = false
		pcall(function()
			valid = validateToken(saved)
		end)
		if valid then
			isActivated = true
			isChecking = false
			screenGui.Enabled = false
			showToast("STRUGLIX", "Auto-activating script...", THEME.success, ASSETS.loading)
			onActivationSuccess()
			return true
		else
			clearSavedToken()
		end
	end
	return false
end

task.spawn(function()
	if tryAutoActivate() then
		return
	end

	screenGui.Enabled = true
	fadeInUI()
	setStatus("[Until]", THEME.textDim, ASSETS.lock, false)
	setButtonState("idle")
end)
